\section{Upgraded Soft-Wall Dynamical System and Global Trapping Box Theorem}

In the unregularized v1.0 system, finite-time escape from $\mathbb{R}^3_+$ occurs through a two-stage boundary breach: (i) when $H_{\text{obs}}$ approaches $0$, the constant erasure term $-\dot{I}_{\text{erase}}$ remains active and drives $H_{\text{obs}} < 0$; and (ii) during large-amplitude oscillations above the subcritical Hopf threshold ($e > e_c$), negative entropy rates $dH_{\text{obs}}/dt < 0$ enter the derivative-feedback term $e\,(dH_{\text{obs}}/dt)$ unattenuated at $\dot{I} = 0$, driving $\dot{I} < 0$ and subsequently pulling $\gamma < 0$ via $a\dot{I}\mathcal{E}_{\text{avail}}(1-\gamma/\gamma_{\max})$.

Note that replacing $-\dot{I}_{\text{erase}}$ with $-\frac{1}{2}\dot{I}_{\text{erase}}[1+\tanh(H_{\text{obs}}/\epsilon_H)]$ or leaving $e\,(dH_{\text{obs}}/dt)$ ungated at $\dot{I}=0$ does \emph{not} render $\mathbb{R}^3_+$ forward-invariant, because $\frac{1}{2}[1+\tanh(0)] = \frac{1}{2} \neq 0$ leaves a negative sink $-\frac{1}{2}\dot{I}_{\text{erase}}$ at $H_{\text{obs}}=0$ whenever $\gamma < \dot{I}_{\text{erase}}/(2cH_{\max})$, and an ungated $e\,(dH_{\text{obs}}/dt)$ drives $\left.\frac{d\dot{I}}{dt}\right|_{\dot{I}=0} < 0$ whenever $e\,(dH_{\text{obs}}/dt) < -f\mathcal{P}_{\text{input}}$.

To construct a modified $C^\infty$ dynamical system in which $\mathbb{R}^3_+$ and a compact box $\mathcal{B}$ are unconditionally forward-invariant, let $\epsilon_H, \epsilon_I > 0$ denote boundary regularization widths ($\epsilon_H \ll H_{\max}$, $\epsilon_I \ll \dot{I}^*$) and define the smooth gates
\begin{equation}
\label{eq:gates}
\Psi_{\epsilon_H}(H_{\text{obs}}) \equiv \tanh\!\left(\frac{H_{\text{obs}}}{\epsilon_H}\right), \qquad \Phi_{\epsilon_I}(\dot{I}) \equiv \tanh\!\left(\frac{\dot{I}}{\epsilon_I}\right).
\end{equation}
The v1.5 soft-wall dynamical system is defined by
\begin{align}
\frac{d\gamma}{dt} &= a \dot{I} \mathcal{E}_{\text{avail}}\!\left(1 - \frac{\gamma}{\gamma_{\max}}\right) - b\gamma, \label{eq:v15_gamma} \\
\frac{dH_{\text{obs}}}{dt} &= c\gamma (H_{\max} - H_{\text{obs}}) - \dot{I}_{\text{erase}}\,\tanh\!\left(\frac{H_{\text{obs}}}{\epsilon_H}\right), \label{eq:v15_H} \\
\frac{d\dot{I}}{dt} &= e\,\tanh\!\left(\frac{\dot{I}}{\epsilon_I}\right)\!\left(\frac{dH_{\text{obs}}}{dt}\right) + f\mathcal{P}_{\text{input}} - g\dot{I}, \label{eq:v15_I}
\end{align}
where all parameters $a, b, c, e, f, g, \mathcal{E}_{\text{avail}}, \mathcal{P}_{\text{input}}, \dot{I}_{\text{erase}}, \gamma_{\max}, H_{\max}, \epsilon_H, \epsilon_I$ are strictly positive.

\begin{theorem}[Global Forward-Invariance of $\mathbb{R}^3_+$ and Compact Trapping Box $\mathcal{B}$]
\label{thm:global_invariance}
For all strictly positive parameters and every feedback gain $e > 0$:
\begin{enumerate}
\setlength{\itemsep}{1pt}\setlength{\parskip}{0pt}
    \item The non-negative orthant $\mathbb{R}^3_+ = \{(\gamma, H_{\text{obs}}, \dot{I}) \in \mathbb{R}^3 \mid \gamma \ge 0,\, H_{\text{obs}} \ge 0,\, \dot{I} \ge 0\}$ is positively invariant under the flow of Eqs.~\eqref{eq:v15_gamma}--\eqref{eq:v15_I}.

    \begin{equation}
    \label{eq:I_max}
    \dot{I}_{\max} \equiv \frac{e M_H + f\mathcal{P}_{\text{input}}}{g} > 0.
    \end{equation}
    Then the compact box $\mathcal{B} \equiv [0, \gamma_{\max}] \times [0, H_{\max}] \times [0, \dot{I}_{\max}] \subset \mathbb{R}^3_+$ is positively invariant and globally absorbing within $[0,\gamma_{\max}]\times[0,H_{\max}]\times[0,\infty)$. Consequently, every trajectory originating in $\mathcal{B}$ exists globally for all $t \in [0,\infty)$ and never escapes to negative states or blows up in finite time.
\end{enumerate}
\end{theorem}

\begin{proof}
Since the vector field $F = (d\gamma/dt, dH_{\text{obs}}/dt, d\dot{I}/dt)$ of Eqs.~\eqref{eq:v15_gamma}--\eqref{eq:v15_I} is $C^\infty$ on $\mathbb{R}^3$, by Nagumo's subtangentiality theorem a closed convex box $\mathcal{B} = \prod_{i=1}^3 [x_{i,\min}, x_{i,\max}]$ is positively invariant if and only if $F_i(x) \ge 0$ whenever $x \in \mathcal{B}$ with $x_i = x_{i,\min}$ and $F_i(x) \le 0$ whenever $x \in \mathcal{B}$ with $x_i = x_{i,\max}$. We check all six boundary faces of $\mathcal{B}$:
\begin{enumerate}
\setlength{\itemsep}{1pt}
    \item \textbf{Face 1 --- Lower $\gamma$-face ($\gamma = 0$, $\dot{I} \ge 0$):} Evaluating Eq.~\eqref{eq:v15_gamma} at $\gamma = 0$ yields
    \begin{equation}
    \left.\frac{d\gamma}{dt}\right|_{\gamma=0} = a\dot{I}\mathcal{E}_{\text{avail}}(1 - 0) - b(0) = a\dot{I}\mathcal{E}_{\text{avail}} \ge 0,
    \end{equation}
    with strict positivity whenever $\dot{I} > 0$.
    \item \textbf{Face 2 --- Upper $\gamma$-ceiling ($\gamma = \gamma_{\max}$, $\dot{I} \ge 0$):} Evaluating Eq.~\eqref{eq:v15_gamma} at $\gamma = \gamma_{\max}$ yields
    \begin{equation}
    \left.\frac{d\gamma}{dt}\right|_{\gamma=\gamma_{\max}} = a\dot{I}\mathcal{E}_{\text{avail}}\!\left(1 - \frac{\gamma_{\max}}{\gamma_{\max}}\right) - b\gamma_{\max} = -b\gamma_{\max} < 0.
    \end{equation}
    \item \textbf{Face 3 --- Lower $H_{\text{obs}}$-face ($H_{\text{obs}} = 0$, $\gamma \ge 0$):} Because $\tanh(0/\epsilon_H) = \tanh(0) = 0$, evaluating Eq.~\eqref{eq:v15_H} at $H_{\text{obs}} = 0$ gives
    \begin{equation}
    \left.\frac{dH_{\text{obs}}}{dt}\right|_{H_{\text{obs}}=0} = c\gamma(H_{\max} - 0) - \dot{I}_{\text{erase}}\tanh(0) = c\gamma H_{\max} \ge 0 \quad \forall \gamma \ge 0.
    \end{equation}
    \item \textbf{Face 4 --- Upper $H_{\text{obs}}$-ceiling ($H_{\text{obs}} = H_{\max}$, $\gamma \ge 0$):} Since $H_{\max}/\epsilon_H > 0 \implies \tanh(H_{\max}/\epsilon_H) > 0$,
    \begin{equation}
    \left.\frac{dH_{\text{obs}}}{dt}\right|_{H_{\text{obs}}=H_{\max}} = c\gamma(0) - \dot{I}_{\text{erase}}\tanh\!\left(\frac{H_{\max}}{\epsilon_H}\right) < 0.
    \end{equation}
    \item \textbf{Face 5 --- Lower $\dot{I}$-face ($\dot{I} = 0$, arbitrary $\gamma, H_{\text{obs}} \in \mathbb{R}$):} Because $\tanh(0/\epsilon_I) = 0$, both the feedback term $e\tanh(0)(dH_{\text{obs}}/dt)$ and the linear damping $-g(0)$ vanish identically at $\dot{I} = 0$:
    \begin{equation}
    \left.\frac{d\dot{I}}{dt}\right|_{\dot{I}=0} = e\cdot 0\cdot \frac{dH_{\text{obs}}}{dt} + f\mathcal{P}_{\text{input}} - g(0) = f\mathcal{P}_{\text{input}} > 0.
    \end{equation}
    \item \textbf{Face 6 --- Upper $\dot{I}$-ceiling ($\dot{I} = \dot{I}_{\max}$ on $\mathcal{B}$):} For any $(\gamma, H_{\text{obs}}) \in [0,\gamma_{\max}]\times[0,H_{\max}]$, $dH_{\text{obs}}/dt \le c\gamma_{\max}H_{\max} \le c\gamma_{\max}H_{\max} + \dot{I}_{\text{erase}} = M_H$. Since $0 \le \tanh(\dot{I}/\epsilon_I) \le 1$, at $\dot{I} = \dot{I}_{\max}$ we have
    \begin{equation}
    \left.\frac{d\dot{I}}{dt}\right|_{\dot{I}=\dot{I}_{\max}} \le e M_H + f\mathcal{P}_{\text{input}} - g\dot{I}_{\max} = 0.
    \end{equation}
\end{enumerate}
Since $F$ points inward on every boundary face of $\mathcal{B}$ and $\mathcal{B}$ is compact, standard ODE global existence guarantees that all trajectories in $\mathcal{B}$ are defined and confined to $\mathcal{B}$ for all $t \ge 0$.
\end{proof}

\begin{enumerate}
\setlength{\itemsep}{1pt}
    \item Since $\left.dH_{\text{obs}}/dt\right|_* = 0$, Eq.~\eqref{eq:v15_I} reduces at steady state to $f\mathcal{P}_{\text{input}} - g\dot{I}^* = 0$ and Eq.~\eqref{eq:v15_gamma} reduces to $K(1-\gamma^*/\gamma_{\max}) - b\gamma^* = 0$ with $K \equiv a\mathcal{E}_{\text{avail}}\dot{I}^*$. Thus $\dot{I}^* = f\mathcal{P}_{\text{input}}/g$ and $\gamma^* = K\gamma_{\max}/(b\gamma_{\max}+K)$ are \textbf{exact} (zero $\epsilon$-dependence), while $H_{\text{obs}}^*$ solves $c\gamma^*(H_{\max}-H_{\text{obs}}^*) = \dot{I}_{\text{erase}}\tanh(H_{\text{obs}}^*/\epsilon_H)$ and differs from $H_{\max} - \dot{I}_{\text{erase}}/(c\gamma^*)$ by $O(e^{-2H_{\text{obs}}^*/\epsilon_H})$.
    \item When linearizing Eq.~\eqref{eq:v15_I} about the equilibrium, the product rule applied to $e\tanh(\dot{I}/\epsilon_I)(dH_{\text{obs}}/dt)$ yields
    \begin{equation}
    \left.\frac{\partial}{\partial \dot{I}}\!\left[e\tanh\!\left(\frac{\dot{I}}{\epsilon_I}\right)\frac{dH_{\text{obs}}}{dt}\right]\right|_* = \frac{e}{\epsilon_I}\operatorname{sech}^2\!\left(\frac{\dot{I}^*}{\epsilon_I}\right)\underbrace{\left.\frac{dH_{\text{obs}}}{dt}\right|_*}_{=\,0} + e\tanh\!\left(\frac{\dot{I}^*}{\epsilon_I}\right)\underbrace{\left.\frac{\partial (dH_{\text{obs}}/dt)}{\partial \dot{I}}\right|_*}_{=\,0} = 0.
    \end{equation}
    Hence the $(3,3)$ entry of the Jacobian is exactly $-g$, with no $\operatorname{sech}^2(\dot{I}^*/\epsilon_I)$ linear damping artifact.
    \item In coordinates $(\gamma, H_{\text{obs}}, \dot{I})$, the exact Jacobian of the upgraded system~\eqref{eq:v15_gamma}--\eqref{eq:v15_I} at equilibrium is
    \begin{equation}
    \label{eq:J_v15}
    J_{\text{v1.5}} = \begin{pmatrix}
    -\alpha & 0 & \beta \\
    \delta_{\epsilon} & -\mu_{\epsilon} & 0 \\
    e_{\text{eff}}\delta_{\epsilon} & -e_{\text{eff}}\mu_{\epsilon} & -g
    \end{pmatrix},
    \end{equation}
    where $\alpha$ and $\beta$ are identical to v1.1--v1.4, $e_{\text{eff}} = e\tanh(\dot{I}^*/\epsilon_I)$, $\delta_{\epsilon} = c(H_{\max}-H_{\text{obs}}^*)$, and $\mu_{\epsilon} = c\gamma^* + \frac{\dot{I}_{\text{erase}}}{\epsilon_H}\operatorname{sech}^2(H_{\text{obs}}^*/\epsilon_H)$. Two distinct exponential gate scales govern the departure from v1.4 as $\epsilon_H, \epsilon_I \to 0^+$:
    \begin{itemize}
    \setlength{\itemsep}{0.5pt}
        \item \textbf{Processing gate scale ($\epsilon_I$):} $1 - \tanh(\dot{I}^*/\epsilon_I) \sim 2e^{-2\dot{I}^*/\epsilon_I}$, which dominates $\|J_{\text{v1.5}} - J_{\text{v1.4}}\|_F = O(e^{-2\dot{I}^*/\epsilon_I})$ and $|e_{c,\text{v1.5}} - e_{c,\text{v1.4}}|$, and via the second and third derivatives of $\tanh(\dot{I}/\epsilon_I)$ shifts $l_1$ by $O(\epsilon_I^{-2}\operatorname{sech}^2(\dot{I}^*/\epsilon_I))$.
        \item \textbf{Entropy gate scale ($\epsilon_H$):} $\operatorname{sech}^2(H_{\text{obs}}^*/\epsilon_H) \sim 4e^{-2H_{\text{obs}}^*/\epsilon_H}$, which shifts $H_{\text{obs}}^*$ and $\mu_\epsilon$ by $O(e^{-2H_{\text{obs}}^*/\epsilon_H})$.
    \end{itemize}
\end{enumerate}
\end{proposition}
