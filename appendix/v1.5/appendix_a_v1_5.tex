Second, by gating the irreversible-erasure sink with $\Psi_{\epsilon_H}(H_{\text{obs}}) = \tanh(H_{\text{obs}}/\epsilon_H)$ and the derivative-feedback channel with $\Phi_{\epsilon_I}(\dot{I}) = \tanh(\dot{I}/\epsilon_I)$, we prove via Nagumo's subtangentiality theorem that both the positive orthant $\mathbb{R}^3_+$ and the compact box $\mathcal{B} = [0,\gamma_{\max}]\times[0,H_{\max}]\times[0,\dot{I}_{\max}]$ are globally forward-invariant for all $t \ge 0$ and all positive parameters. \section{Conditional Holographic Dictionary for Gate 0}

In \textit{Physics v1.0}, the observer von Neumann entropy $H_{\text{obs}}(t) = -\operatorname{Tr}(\rho_{\text{obs}}\log_2 \rho_{\text{obs}})$ is bounded by an informational saturation capacity $H_{\max}$ (in bits). The Gate~0 Technical Report recorded the bridge between an abstract subsystem factorization $\mathcal{H}_{\text{total}} = \mathcal{H}_{\text{obs}} \otimes \mathcal{H}_{\text{env}}$ and geometric horizon entropy $A/(4\ell_P^2\ln 2)$ as \textbf{UNRESOLVED IN GENERALITY} because an arbitrary algebraic tensor factorization need not correspond to a spatially localized region with a well-defined boundary area.

To make the microphysical assumption required to close Gate~0 mathematically explicit, we replace static Ryu--Takayanagi identification with a covariant causal-diamond localization postulate:

\begin{proposition}[Conditional Holographic Closure of Gate 0]
Let $(\mathcal{M}, g_{\mu\nu})$ be a $(3+1)$-dimensional semiclassical spacetime, and suppose the observer subsystem $\mathcal{H}_{\text{obs}}$ is defined by the algebra of quantum fields localized within the causal diamond $\mathcal{D}(\mathcal{A})$ of a spacelike ball $\mathcal{A}$ with smooth codimension-two boundary $\partial\mathcal{A}$. Under the covariant Bousso entropy bound and the Hubeny--Rangamani--Takayanagi (HRT) extremal surface prescription, if $\Sigma_{\text{HRT}}$ is the unique minimal-area extremal surface homologous to $\mathcal{A}$ with $\partial\Sigma_{\text{HRT}} = \partial\mathcal{A}$ and stationary area $\operatorname{Area}(\Sigma_{\text{HRT}}) = A_{\Sigma}$, then the state-space supremum of the observer's fine-grained von Neumann entropy in bits is fixed by
\begin{equation}
\label{eq:hrt_dictionary}
H_{\max} \equiv \sup_{\rho_{\text{obs}} \in \mathcal{S}(\mathcal{D}(\mathcal{A}))} \left[-\operatorname{Tr}\!\left(\rho_{\text{obs}}\log_2 \rho_{\text{obs}}\right)\right] = \frac{\operatorname{Area}(\Sigma_{\text{HRT}})}{4\ell_P^2 \ln 2} \quad [\text{bits}].
\end{equation}
\end{proposition}

\begin{remark}[Distinction Between Conditional Closure and General Derivation]
Equation~\eqref{eq:hrt_dictionary} closes Gate~0 \emph{conditionally} within the class of spatially localized causal-diamond subsystems admitting a semiclassical holographic or covariant entropy bound. For abstract, non-geometric Hilbert-space factorizations $\mathcal{H}_{\text{total}} = \mathcal{H}_{\text{obs}} \otimes \mathcal{H}_{\text{env}}$ without a spatial boundary $\partial\mathcal{A}$, Gate~0 remains open. None of the dynamical-sector theorems in Sections 3--6 depend on Eq.~\eqref{eq:hrt_dictionary}; they hold for any constant $H_{\max} > 0$.
\end{remark}

\section{Upgraded Soft-Wall Dynamical System and Global Trapping Box Theorem}

In the unregularized v1.0 system, finite-time escape from $\mathbb{R}^3_+$ occurs through a two-stage boundary breach: (i) when $H_{\text{obs}}$ approaches $0$, the constant erasure term $-\dot{I}_{\text{erase}}$ remains active and drives $H_{\text{obs}} < 0$; and (ii) during large-amplitude oscillations above the subcritical Hopf threshold ($e > e_c$), negative entropy rates $dH_{\text{obs}}/dt < 0$ enter the derivative-feedback term $e\,(dH_{\text{obs}}/dt)$ unattenuated at $\dot{I} = 0$, driving $\dot{I} < 0$ and subsequently pulling $\gamma < 0$ via $a\dot{I}\mathcal{E}_{\text{avail}}(1-\gamma/\gamma_{\max})$.

Note that replacing $-\dot{I}_{\text{erase}}$ with $-\frac{1}{2}\dot{I}_{\text{erase}}[1+\tanh(H_{\text{obs}}/\epsilon_H)]$ or leaving $e\,(dH_{\text{obs}}/dt)$ ungated at $\dot{I}=0$ does \emph{not} render $\mathbb{R}^3_+$ forward-invariant, because $\frac{1}{2}[1+\tanh(0)] = \frac{1}{2} \neq 0$ leaves a negative sink $-\frac{1}{2}\dot{I}_{\text{erase}}$ at $H_{\text{obs}}=0$ whenever $\gamma < \dot{I}_{\text{erase}}/(2cH_{\max})$, and an ungated $e\,(dH_{\text{obs}}/dt)$ drives $\left.\frac{d\dot{I}}{dt}\right|_{\dot{I}=0} < 0$ whenever $e\,(dH_{\text{obs}}/dt) < -f\mathcal{P}_{\text{input}}$.

To achieve unconditional global forward-invariance of $\mathbb{R}^3_+$ via smooth ($C^\infty$) real-analytic multipliers that vanish at $H_{\text{obs}}=0$ and $\dot{I}=0$ and saturate to $1$ in the interior, let $\epsilon_H, \epsilon_I > 0$ denote micro-scale boundary regularization widths ($\epsilon_H \ll H_{\max}$, $\epsilon_I \ll \dot{I}^*$) and define the gates
\begin{equation}
\label{eq:gates}
\Psi_{\epsilon_H}(H_{\text{obs}}) \equiv \tanh\!\left(\frac{H_{\text{obs}}}{\epsilon_H}\right), \qquad \Phi_{\epsilon_I}(\dot{I}) \equiv \tanh\!\left(\frac{\dot{I}}{\epsilon_I}\right).
\end{equation}
The upgraded v1.5 dynamical system is governed by
\begin{align}
\frac{d\gamma}{dt} &= a \dot{I} \mathcal{E}_{\text{avail}}\!\left(1 - \frac{\gamma}{\gamma_{\max}}\right) - b\gamma, \label{eq:v15_gamma} \\
\frac{dH_{\text{obs}}}{dt} &= c\gamma (H_{\max} - H_{\text{obs}}) - \dot{I}_{\text{erase}}\,\tanh\!\left(\frac{H_{\text{obs}}}{\epsilon_H}\right), \label{eq:v15_H} \\
\frac{d\dot{I}}{dt} &= e\,\tanh\!\left(\frac{\dot{I}}{\epsilon_I}\right)\!\left(\frac{dH_{\text{obs}}}{dt}\right) + f\mathcal{P}_{\text{input}} - g\dot{I}, \label{eq:v15_I}
\end{align}
where all parameters $a, b, c, e, f, g, \mathcal{E}_{\text{avail}}, \mathcal{P}_{\text{input}}, \dot{I}_{\text{erase}}, \gamma_{\max}, H_{\max}, \epsilon_H, \epsilon_I$ are strictly positive.

\begin{theorem}[Global Forward-Invariance of $\mathbb{R}^3_+$ and Compact Trapping Box $\mathcal{B}$]
\label{thm:global_invariance}
For all strictly positive parameters and every feedback gain $e > 0$:
\begin{enumerate}
\setlength{\itemsep}{2pt}\setlength{\parskip}{0pt}
    \item The non-negative orthant $\mathbb{R}^3_+ = \{(\gamma, H_{\text{obs}}, \dot{I}) \in \mathbb{R}^3 \mid \gamma \ge 0,\, H_{\text{obs}} \ge 0,\, \dot{I} \ge 0\}$ is positively invariant under the flow of Eqs.~\eqref{eq:v15_gamma}--\eqref{eq:v15_I}.
    \item Define the maximum entropy-rate bound $M_H \equiv c\gamma_{\max}H_{\max} + \dot{I}_{\text{erase}}$ and the processing ceiling
    \begin{equation}
    \label{eq:I_max}
    \dot{I}_{\max} \equiv \frac{e M_H + f\mathcal{P}_{\text{input}}}{g} > 0.
    \end{equation}
    Then the compact box $\mathcal{B} \equiv [0, \gamma_{\max}] \times [0, H_{\max}] \times [0, \dot{I}_{\max}] \subset \mathbb{R}^3_+$ is positively invariant and globally absorbing within $[0,\gamma_{\max}]\times[0,H_{\max}]\times[0,\infty)$. Consequently, every trajectory originating in $\mathcal{B}$ exists globally for all $t \in [0,\infty)$ and never escapes to unphysical negative states or blows up in finite time.
\end{enumerate}
\end{theorem}

\begin{proof}
Since the vector field $F = (d\gamma/dt, dH_{\text{obs}}/dt, d\dot{I}/dt)$ of Eqs.~\eqref{eq:v15_gamma}--\eqref{eq:v15_I} is $C^\infty$ (and therefore locally Lipschitz) on $\mathbb{R}^3$, by Nagumo's subtangentiality theorem a closed convex box $\mathcal{B} = \prod_{i=1}^3 [x_{i,\min}, x_{i,\max}]$ is positively invariant if and only if $F_i(x) \ge 0$ whenever $x \in \mathcal{B}$ with $x_i = x_{i,\min}$ and $F_i(x) \le 0$ whenever $x \in \mathcal{B}$ with $x_i = x_{i,\max}$. We check all six boundary faces of $\mathcal{B}$ (and the three lower faces of $\mathbb{R}^3_+$):
\begin{enumerate}
\setlength{\itemsep}{2pt}
    \item \textbf{Face 1 --- Lower $\gamma$-face ($\gamma = 0$, $\dot{I} \ge 0$):} Evaluating Eq.~\eqref{eq:v15_gamma} at $\gamma = 0$ yields
    \begin{equation}
    \left.\frac{d\gamma}{dt}\right|_{\gamma=0} = a\dot{I}\mathcal{E}_{\text{avail}}(1 - 0) - b(0) = a\dot{I}\mathcal{E}_{\text{avail}} \ge 0,
    \end{equation}
    with strict positivity whenever $\dot{I} > 0$.
    \item \textbf{Face 2 --- Upper $\gamma$-ceiling ($\gamma = \gamma_{\max}$, $\dot{I} \ge 0$):} Evaluating Eq.~\eqref{eq:v15_gamma} at $\gamma = \gamma_{\max}$ yields
    \begin{equation}
    \left.\frac{d\gamma}{dt}\right|_{\gamma=\gamma_{\max}} = a\dot{I}\mathcal{E}_{\text{avail}}\!\left(1 - \frac{\gamma_{\max}}{\gamma_{\max}}\right) - b\gamma_{\max} = -b\gamma_{\max} < 0.
    \end{equation}
    \item \textbf{Face 3 --- Lower $H_{\text{obs}}$-face ($H_{\text{obs}} = 0$, $\gamma \ge 0$):} Because $\tanh(0/\epsilon_H) = \tanh(0) = 0$, evaluating Eq.~\eqref{eq:v15_H} at $H_{\text{obs}} = 0$ gives
    \begin{equation}
    \left.\frac{dH_{\text{obs}}}{dt}\right|_{H_{\text{obs}}=0} = c\gamma(H_{\max} - 0) - \dot{I}_{\text{erase}}\tanh(0) = c\gamma H_{\max} \ge 0 \quad \forall \gamma \ge 0.
    \end{equation}
    Notably, this non-negativity holds unconditionally for every $\dot{I}_{\text{erase}} > 0$ and every $\gamma \ge 0$, without requiring $\gamma(t) \ge \gamma^*$.
    \item \textbf{Face 4 --- Upper $H_{\text{obs}}$-ceiling ($H_{\text{obs}} = H_{\max}$, $\gamma \ge 0$):} Since $H_{\max}/\epsilon_H > 0 \implies \tanh(H_{\max}/\epsilon_H) > 0$,
    \begin{equation}
    \left.\frac{dH_{\text{obs}}}{dt}\right|_{H_{\text{obs}}=H_{\max}} = c\gamma(0) - \dot{I}_{\text{erase}}\tanh\!\left(\frac{H_{\max}}{\epsilon_H}\right) < 0.
    \end{equation}
    \item \textbf{Face 5 --- Lower $\dot{I}$-face ($\dot{I} = 0$, arbitrary $\gamma, H_{\text{obs}} \in \mathbb{R}$):} Because $\tanh(0/\epsilon_I) = 0$, both the feedback term $e\tanh(0)(dH_{\text{obs}}/dt)$ and the linear damping $-g(0)$ vanish identically at $\dot{I} = 0$:
    \begin{equation}
    \left.\frac{d\dot{I}}{dt}\right|_{\dot{I}=0} = e\cdot 0\cdot \frac{dH_{\text{obs}}}{dt} + f\mathcal{P}_{\text{input}} - g(0) = f\mathcal{P}_{\text{input}} > 0.
    \end{equation}
    \item \textbf{Face 6 --- Upper $\dot{I}$-ceiling ($\dot{I} = \dot{I}_{\max}$ on $\mathcal{B}$):} For any $(\gamma, H_{\text{obs}}) \in [0,\gamma_{\max}]\times[0,H_{\max}]$, $dH_{\text{obs}}/dt \le c\gamma_{\max}H_{\max} \le c\gamma_{\max}H_{\max} + \dot{I}_{\text{erase}} = M_H$. Since $0 \le \tanh(\dot{I}/\epsilon_I) \le 1$, at $\dot{I} = \dot{I}_{\max}$ we have
    \begin{equation}
    \left.\frac{d\dot{I}}{dt}\right|_{\dot{I}=\dot{I}_{\max}} \le e M_H + f\mathcal{P}_{\text{input}} - g\dot{I}_{\max} = 0.
    \end{equation}
\end{enumerate}
Since $F$ points inward on every boundary face of $\mathcal{B}$ and $\mathcal{B}$ is compact, standard ODE global existence guarantees that all trajectories in $\mathcal{B}$ are defined and confined to $\mathcal{B}$ for all $t \ge 0$.
\end{proof}

\takeaway{Key Structural Insight}{In the post-Hopf unstable regime ($e > e_c$), the unique interior equilibrium loses stability via a subcritical Hopf bifurcation ($l_1 > 0$), while the boundary of the compact box $\mathcal{B} = [0,\gamma_{\max}]\times[0,H_{\max}]\times[0,\dot{I}_{\max}]$ is strictly inward-pointing. Instead of escaping to $-\infty$ in finite time as in v1.0, trajectories repelled from the equilibrium are trapped inside $\mathcal{B}$ and settle onto a globally bounded relaxation oscillation.}

\section{Exact Equilibrium and Multi-Scale Spectral Invariance in the Interior}

A critical advantage of gating only the erasure sink $\dot{I}_{\text{erase}}\tanh(H_{\text{obs}}/\epsilon_H)$ and the derivative-feedback term $e\tanh(\dot{I}/\epsilon_I)(dH_{\text{obs}}/dt)$---while keeping the linear relaxation terms $-b\gamma$ and $-g\dot{I}$ unperturbed---is that the interior steady-state coordinates $(\gamma^*, \dot{I}^*)$ are exact and the first-order gate derivative vanishes identically at equilibrium:

\begin{proposition}[Exactness of $(\gamma^*,\dot{I}^*)$ and Multi-Scale Gate Perturbation Bounds]
\label{prop:jacobian_invariance}
At any equilibrium $(\gamma^*, H_{\text{obs}}^*, \dot{I}^*)$ of Eqs.~\eqref{eq:v15_gamma}--\eqref{eq:v15_I}:
\begin{enumerate}
\setlength{\itemsep}{2pt}
    \item Since $\left.dH_{\text{obs}}/dt\right|_* = 0$, Eq.~\eqref{eq:v15_I} reduces at steady state to $f\mathcal{P}_{\text{input}} - g\dot{I}^* = 0$ and Eq.~\eqref{eq:v15_gamma} reduces to $K(1-\gamma^*/\gamma_{\max}) - b\gamma^* = 0$ with $K \equiv a\mathcal{E}_{\text{avail}}\dot{I}^*$. Thus $\dot{I}^* = f\mathcal{P}_{\text{input}}/g$ and $\gamma^* = K\gamma_{\max}/(b\gamma_{\max}+K)$ are \textbf{identical} to Proposition~2 of Appendix~A v1.1 with zero $\epsilon$-dependence.
    \item When linearizing Eq.~\eqref{eq:v15_I} about the equilibrium, the product rule applied to $e\tanh(\dot{I}/\epsilon_I)(dH_{\text{obs}}/dt)$ yields
    \begin{equation}
    \left.\frac{\partial}{\partial \dot{I}}\!\left[e\tanh\!\left(\frac{\dot{I}}{\epsilon_I}\right)\frac{dH_{\text{obs}}}{dt}\right]\right|_* = \frac{e}{\epsilon_I}\operatorname{sech}^2\!\left(\frac{\dot{I}^*}{\epsilon_I}\right)\underbrace{\left.\frac{dH_{\text{obs}}}{dt}\right|_*}_{=\,0} + e\tanh\!\left(\frac{\dot{I}^*}{\epsilon_I}\right)\underbrace{\left.\frac{\partial (dH_{\text{obs}}/dt)}{\partial \dot{I}}\right|_*}_{=\,0} = 0.
    \end{equation}
    Hence the $(3,3)$ entry of the Jacobian is exactly $-g$, with no $\operatorname{sech}^2(\dot{I}^*/\epsilon_I)$ linear damping artifact.
    \item In coordinates $(\gamma, H_{\text{obs}}, \dot{I})$, the exact Jacobian of the upgraded system~\eqref{eq:v15_gamma}--\eqref{eq:v15_I} at equilibrium is
    \begin{equation}
    \label{eq:J_v15}
    J_{\text{v1.5}} = \begin{pmatrix}
    -\alpha & 0 & \beta \\
    \delta_{\epsilon} & -\mu_{\epsilon} & 0 \\
    e_{\text{eff}}\delta_{\epsilon} & -e_{\text{eff}}\mu_{\epsilon} & -g
    \end{pmatrix},
    \end{equation}
    where $\alpha$ and $\beta$ are identical to v1.1--v1.4, $e_{\text{eff}} = e\tanh(\dot{I}^*/\epsilon_I)$, $\delta_{\epsilon} = c(H_{\max}-H_{\text{obs}}^*)$, and $\mu_{\epsilon} = c\gamma^* + \frac{\dot{I}_{\text{erase}}}{\epsilon_H}\operatorname{sech}^2(H_{\text{obs}}^*/\epsilon_H)$. Two distinct exponential gate scales govern the departure from v1.4:
    \begin{itemize}
    \setlength{\itemsep}{1pt}
        \item \textbf{Processing gate scale ($\epsilon_I$):} $1 - \tanh(\dot{I}^*/\epsilon_I) \sim 2e^{-2\dot{I}^*/\epsilon_I}$, which dominates the Frobenius norm difference $\|J_{\text{v1.5}} - J_{\text{v1.4}}\|_F = O(e^{-2\dot{I}^*/\epsilon_I})$ ($1.48 \times 10^{-86}$ at $\dot{I}^*=1, \epsilon_I=0.01$) and, via the second and third derivatives of $\tanh(\dot{I}/\epsilon_I)$ in the normal-form bilinear/trilinear contractions, shifts $l_1$ by $O(\epsilon_I^{-2}\operatorname{sech}^2(\dot{I}^*/\epsilon_I)) = 1.11 \times 10^{-82}$.
        \item \textbf{Entropy gate scale ($\epsilon_H$):} $\operatorname{sech}^2(H_{\text{obs}}^*/\epsilon_H) \sim 4e^{-2H_{\text{obs}}^*/\epsilon_H}$, which shifts $H_{\text{obs}}^*$ and $\mu_\epsilon$ by $O(e^{-2H_{\text{obs}}^*/\epsilon_H})$ ($5.38 \times 10^{-695}$ at $H_{\text{obs}}^*=8, \epsilon_H=0.01$).
    \end{itemize}
\end{enumerate}
\end{proposition}
